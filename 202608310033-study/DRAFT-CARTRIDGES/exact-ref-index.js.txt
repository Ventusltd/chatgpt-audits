/* ==========================================================================
 * BLOCK 3 — the index loader and the fast-path resolver
 * ANCHOR:
 *   async function runtime() {
 * REPLACEMENT: the following block, THEN the anchor.

  // ---- 2026xxxx exact-ref index lane ----------------------------------------
  // A minimal repd_ref -> point map. Seven fields, ~1.05 MB raw / ~340 KB gzip
  // for all 11,069 official records. It is NOT a search index: it answers
  // exactly one question, "where is this ref", which is the only question a
  // deep link asks. Free-text search remains the Parquet/DuckDB lane below.

  let refIndexPromise = null;

  async function loadRefIndex() {
    refIndexPromise ||= (async () => {
      const started = performance.now();
      const response = await fetch(REF_INDEX_URL, { cache: 'force-cache' });
      invariant(response.ok, `REPD ref index HTTP ${response.status}`);
      const bytes = await response.arrayBuffer();
      invariant(await sha256Hex(bytes) === REF_INDEX_SHA256, 'REPD ref index SHA-256 mismatch');
      const payload = JSON.parse(new TextDecoder().decode(bytes));
      invariant(payload?.schema === 'gridatlas.repd-ref-index.v1', 'REPD ref index schema mismatch');
      invariant(payload?.generation === '202608290716', 'REPD ref index generation mismatch');
      invariant(payload?.rows === REF_INDEX_ROWS, 'REPD ref index row closure mismatch');
      // Same Parquet identity the search lane verifies, so the two lanes can
      // never silently describe different universes.
      invariant(payload?.parquet_sha256 === PARQUET_SHA256, 'REPD ref index Parquet identity mismatch');
      invariant(payload?.records && typeof payload.records === 'object', 'REPD ref index records missing');
      state.ref_index.loaded = true;
      state.ref_index.verified = true;
      state.ref_index.rows = payload.rows;
      state.ref_index.ms = Math.round(performance.now() - started);
      return payload.records;
    })();
    return refIndexPromise;
  }

  // Returns the same shape queryOfficialRepd() returns, so selectResult() and
  // renderResults() need no change at all.
  async function lookupExactRef(repdRef) {
    const records = await loadRefIndex();
    const row = records[String(repdRef)];
    if (!row) return null;
    // Index row order is fixed by the builder and asserted by its own test:
    // [name, technology, status, capacity_mw, longitude, latitude]
    const [name, technology, status, capacityMw, longitude, latitude] = row;
    return {
      repd_ref: String(repdRef),
      name: String(name ?? ''),
      address: '',
      postcode: '',
      county: '',
      planning_authority: '',
      technology: String(technology ?? ''),
      status: String(status ?? ''),
      capacity_mw: capacityMw == null ? null : Number(capacityMw),
      longitude: longitude == null ? null : Number(longitude),
      latitude: latitude == null ? null : Number(latitude),
      score: 10000,
      lane: 'REF_INDEX'
    };
  }


/* ==========================================================================
 * BLOCK 4 — the deep-link receiver, rewritten
 * ANCHOR: the whole existing function, from
 *   async function receiveExactRepdDeepLink(input, resultsEl) {
 * to its closing brace. REPLACEMENT:
 * ==========================================================================
 *
 * CHANGES vs the parent:
 *  - resolves from the ref index first; DuckDB is never booted for a deep link
 *  - splits the outcome into four classes instead of resolved/failed, so a
 *    project with no safe map point is RESOLVED_NOT_MAPPED, not a failure
 *    (28 spine projects have geometry_status=missing; see intelligence-chain
 *    section 5.1)
 *  - BUT keeps status === 'RESOLVED' for the mapped case, because
 *    atman/202608301624-verify-v9-5-search.mjs waits on exactly that string
 *    (line: deep_link?.status === 'RESOLVED'). The new information is carried
 *    in a NEW resolution_class field, so the existing gate passes unchanged.
 *  - falls back to the Parquet lane only if the index is unavailable, so a
 *    stale cache or a CDN failure degrades rather than breaks
 * ========================================================================== */

  async function receiveExactRepdDeepLink(input, resultsEl) {
    const repdRef = String(new URLSearchParams(window.location.search).get('repd_ref') || '').trim();
    if (!repdRef) {
      state.deep_link = { status: 'ABSENT', repd_ref: null, resolved: false, mapped: false };
      document.body.dataset.gridatlasRepdDeepLink = 'absent';
      return;
    }

    state.deep_link = { status: 'RECEIVING', repd_ref: repdRef, resolved: false, mapped: false };
    try {
      if (!/^[A-Za-z0-9-]{1,40}$/.test(repdRef)) {
        state.deep_link = { status: 'MALFORMED', repd_ref: repdRef, resolved: false, mapped: false };
        document.body.dataset.gridatlasRepdDeepLink = 'malformed';
        return;
      }

      input.value = repdRef;

      // FAST PATH: index first, DuckDB never touched.
      let exact = null;
      try {
        exact = await lookupExactRef(repdRef);
      } catch (indexError) {
        state.failures.push({ phase: 'ref_index', message: String(indexError?.message || indexError) });
      }

      // FALLBACK: the original Parquet lane, only if the index failed.
      if (!exact) {
        const results = await queryOfficialRepd(repdRef);
        state.duckdb_booted_for = 'deep_link_fallback';
        exact = results.find(result => String(result.repd_ref) === repdRef) || null;
        if (exact) renderResults(results, resultsEl);
      }

      if (!exact) {
        state.deep_link = { status: 'NOT_FOUND', repd_ref: repdRef, resolved: false, mapped: false };
        document.body.dataset.gridatlasRepdDeepLink = 'not-found';
        return;
      }

      await waitForCapturedMap();
      selectResult(exact);
      invariant(state.last_selection?.repd_ref === repdRef, 'exact REPD selection was not retained');

      const mapped = state.last_selection?.mapped === true;
      document.body.dataset.gridatlasRepdRef = repdRef;
      document.body.dataset.gridatlasRepdDeepLink = mapped ? 'resolved' : 'resolved-unmapped';
      state.deep_link = {
        // status keeps the parent vocabulary so the existing browser gate,
        // which waits on the literal string 'RESOLVED', is unaffected.
        status: mapped ? 'RESOLVED' : 'RESOLVED_NOT_MAPPED',
        resolution_class: mapped ? 'RESOLVED_AND_MAPPED' : 'RESOLVED_NOT_MAPPED',
        repd_ref: repdRef,
        resolved: true,
        mapped,
        lane: exact.lane || 'PARQUET',
        name: exact.name,
        postcode: exact.postcode,
        longitude: exact.longitude,
        latitude: exact.latitude
      };
    } catch (error) {
      const message = String(error?.message || error);
      state.failures.push({ phase: 'exact_repd_deep_link', repd_ref: repdRef, message });
      state.deep_link = { status: 'FAILED', repd_ref: repdRef, resolved: false, mapped: false, message };
      document.body.dataset.gridatlasRepdDeepLink = 'failed';
      console.error('[V9 EXACT REPD DEEP LINK]', error);
    }
  }
